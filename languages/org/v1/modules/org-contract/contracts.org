* Evidence link contract
:PROPERTIES:
:CONTRACT_ID: section.scope.v1
:CONTRACT_SCOPE: subtree
:END:

#+name: evidence-link-assertions
#+begin_src scheme :org-contract

  (bindings
   (bind evidence
     (org-elements headline (property title "Evidence")

  (expect at-least 1))
